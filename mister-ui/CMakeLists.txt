cmake_minimum_required(VERSION 3.20)
project(mister-ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
add_executable(mister-ui main.cpp)
target_link_libraries(mister-ui PRIVATE Qt6::Widgets Qt6::Network)
set_target_properties(mister-ui PROPERTIES OUTPUT_NAME misterMPV)
